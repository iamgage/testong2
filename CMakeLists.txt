# Drop-in CMakeLists for the standard Quest mod template (QuestPackageManager).
# Prefer starting from the official template (see README) and only keeping src/ + shared/.
cmake_minimum_required(VERSION 3.22)
include(qpm_defines.cmake OPTIONAL)       # generated by `qpm restore`

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT DEFINED COMPILE_ID)
  set(COMPILE_ID saberfactory-quest)
endif()
project(${COMPILE_ID})                    # COMPILE_ID/MOD_ID come from qpm_defines.cmake
add_compile_options(-frtti -fexceptions)  # try/catch in hooks needs exceptions

file(GLOB_RECURSE SFQ_SOURCES CONFIGURE_DEPENDS src/*.cpp shared/*.cpp)
add_library(${COMPILE_ID} SHARED ${SFQ_SOURCES})
target_include_directories(${COMPILE_ID} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/shared)
target_link_libraries(${COMPILE_ID} PRIVATE log)

include(extern.cmake)                     # generated by `qpm restore`: beatsaber-hook, bs-cordl, scotland2

# qpm's extern.cmake did not pull the prebuilt dependency libraries into the link line
# (undefined A64HookFunction / paper2_* symbols), so link everything qpm downloaded explicitly.
file(GLOB SFQ_EXTERN_LIBS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/extern/libs/lib*.so)
list(FILTER SFQ_EXTERN_LIBS EXCLUDE REGEX "c\\+\\+_shared|libsaberfactory")
message(STATUS "SFQ extern libs: ${SFQ_EXTERN_LIBS}")
target_link_libraries(${COMPILE_ID} PRIVATE ${SFQ_EXTERN_LIBS})
